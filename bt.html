아래에 추가할 버튼 ── /* */ // ── wRSI 계산 (주간 RSI) ── function calcWeeklyRsi(bars, endIdx) { // 주간 종가 맵 생성 const weekly = {}; for (let i = 0; i <= endIdx; i++) { const d = new Date(bars[i].date); const day = d.getDay(); const monday = new Date(d); monday.setDate(d.getDate() - (day === 0 ? 100 : 100 - 100 / (1 + ag / al); } if (wcloses.length < 15) return { wrsi: 50, prevWrsi: 50 }; return { wrsi: rsi14(wcloses), prevWrsi: rsi14(wcloses.slice(0, -1)) }; } // ── 메인 시뮬레이션 ── function simulateNewSystem(bars, userStartIdx) { const TU = 20; const closes = bars.map(b => b.close); const dates = bars.map(b => b.date); const n = closes.length; const warmup = Math.max(userStartIdx, 25); let SEED = 100000; let cash = SEED; let pos = []; // {bp, sh, bi, zone} (attack only) let defPos = []; // {bp, sh, bi, ma0} (defense) let peak = SEED; let mdd = 0; let nBuys = 0, nSells = 0, W = 0, L = 0; // wRSI 모드 캐시 (날짜별) const modeCache = {}; let cachedMode = 'attack'; function getMode(i) { const key = dates[i].slice(0, 10); if (modeCache[key] !== undefined) return modeCache[key]; const { wrsi, prevWrsi } = calcWeeklyRsi(bars, i); const rising = wrsi > prevWrsi; let mode = cachedMode; if (prevWrsi < 50 && wrsi >= 50) mode = 'attack'; else if (rising && wrsi >= 50 && wrsi <= 60) mode = 'attack'; else if (rising && wrsi < 35) mode = 'attack'; else if (prevWrsi >= 50 && wrsi < 50) mode = 'defense'; else if (!rising && wrsi >= 40 && wrsi <= 50) mode = 'defense'; else if (!rising && wrsi > 65) mode = 'defense'; cachedMode = mode; modeCache[key] = mode; return mode; } // MA3 function ma3(i) { const s = Math.max(0, i - 2); let sum = 0, cnt = 0; for (let k = s; k <= i; k++) { sum += closes[k]; cnt++; } return sum / cnt; } // high20 function high20(i) { return Math.max(...closes.slice(Math.max(0, i - 19), i + 1)); } // 연/월별 스냅샷 const periodSnapshots = {}; let lastYear = '', lastMonth = ''; function snapTV(i) { let v = cash; pos.forEach(p => v += p.sh * closes[i]); defPos.forEach(p => v += p.sh * closes[i]); return v; } // 사이클 추적 const cycles = []; let inCycle = false, cycleStartCap = SEED, cycleStartIdx = warmup; for (let i = warmup; i < n; i++) { const p = closes[i]; const pv = closes[i - 1]; const fi = p > pv; // 오늘 상승일? completed.reduce((s, c) => s + c.pnl, 0) / completed.length : 0; const avgDur = completed.length ? completed.reduce((s, c) => s + c.duration, 0) / completed.length : 0; return { mult: final / 100000, mdd, nBuys, nSells, W, L, cycles, cycleTotal: cycles.length, cycleWins, cycleLosses: completed.length - cycleWins, cycleWinRate: completed.length ? cycleWins / completed.length * 100 : 0, avgCyclePnl: avgPnl, avgCycleDuration: avgDur, periodSnapshots, incompleteCount: inCycle ? 1 : 0 }; } // ── 버튼 핸들러 ── async function runNewSystem() { const sym = document.getElementById('bt-sym').value.toUpperCase().trim(); const userStartDate = document.getElementById('bt-start').value; const userEndDate = document.getElementById('bt-end').value; if (!sym) { alert('종목 입력!'); return; } if (!userStartDate || !userEndDate){ alert('날짜 선택!'); return; } const progressEl = document.getElementById('bt-progress'); const resultEl = document.getElementById('result'); progressEl.innerHTML = '🆕 NEW SYSTEM 백테 중...'; resultEl.innerHTML = ''; try { // warmup 포함 데이터 fetch const fetchStart = new Date(userStartDate); fetchStart.setDate(fetchStart.getDate() - 60); const fmt = d => d.toISOString().slice(0, 10); const res = await fetch(`/api/history?symbol=${sym}&start=${fmt(fetchStart)}&end=${userEndDate}`); const data = await res.json(); if (data.error) { progressEl.innerHTML = '❌ ' + data.error; return; } if (!data.bars || data.bars.length < 30) { progressEl.innerHTML = '❌ 데이터 부족'; return; } const userStartTs = new Date(userStartDate).getTime(); let userStartIdx = 0; for (let i = 0; i < data.bars.length; i++) { if (new Date(data.bars[i].date).getTime() >= userStartTs) { userStartIdx = i; break; } } // 기본 HUNTER 비교용 const paramsBasic = { top_p:0.025, mid_p:0.005, bot_p:0.03, top_d:7, mid_d:7, bot_d:10, lookback:20, up:2, dn:4 }; const hunterR = simulateHunter(data.bars, paramsBasic, userStartIdx, 0); const newR = simulateNewSystem(data.bars, userStartIdx); const bh = data.bars[data.bars.length - 1].close / data.bars[userStartIdx].close; const totalDays = data.bars.length - userStartIdx; const years = totalDays / 252; function cagr(mult) { return (Math.pow(mult, 1 / years) - 1) * 100; } function calmar(mult, mddPct) { return Math.abs(mddPct) > 0 ?
